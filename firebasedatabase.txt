rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // --- HELPERS ---
    function isAuthenticated() {
      return request.auth != null;
    }
    function isOwner(userId) {
      return request.auth.uid == userId;
    }
    function isAdmin() {
      return isAuthenticated() && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    // --- COLLECTIONS ---

    // 1. USERS: Admins can do anything; Owners can only update non-role fields
    match /users/{userId} {
      allow read: if isAuthenticated();
      
      allow create: if isOwner(userId);
      
      // FIX: Allow update if (Owner AND not changing role) OR (Admin)
      allow update: if (isOwner(userId) && request.resource.data.role == resource.data.role) || isAdmin();
      
      // FIX: Allow delete if Admin
      allow delete: if isAdmin();
    }

    // 2. ADMINS: Admins can promote/demote (write), Users can read their own status
    match /admins/{adminId} {
      allow read: if request.auth.uid == adminId;
      allow write: if isAdmin();
    }

    // 3. SETTINGS: Admins can manage site settings
    match /settings/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // 4. COURSES: Public read, Admin write
    match /courses/{courseId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
      
      match /lectures/{lectureId} {
        allow read: if true;
        allow write: if isAdmin();
      }
      
      match /doubts/{doubtId} {
        allow read: if isAuthenticated();
        allow create: if isAuthenticated();
        allow update, delete: if isOwner(resource.data.userId) || isAdmin();
      }
    }
    
    // Default Deny
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
